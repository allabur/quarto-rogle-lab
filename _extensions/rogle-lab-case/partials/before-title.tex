\newcommand{\quartoFooterLeft}{%
  \small
  $if(license)$
  $if(license.logo)$
    $if(license.url)$
      \href{$license.url$}{\raisebox{-0.3\height}{\includegraphics[height=1.2em]{$license.logo$}}}\hspace{0.5em}%
    $else$
      \raisebox{-0.3\height}{\includegraphics[height=1.2em]{$license.logo$}}\hspace{0.5em}%
    $endif$
  $else$
    \ifthenelse{\equal{\quartoLicenseLogoPath}{}}{%
      \quartoLicenseTextFallback%
    }{%
      \IfFileExists{\quartoLicenseLogoPath}{%
        \ifthenelse{\equal{\quartoLicenseUrl}{}}{%
          \raisebox{-0.3\height}{\includegraphics[height=1.2em]{\quartoLicenseLogoPath}}%
        }{%
          \href{\quartoLicenseUrl}{\raisebox{-0.3\height}{\includegraphics[height=1.2em]{\quartoLicenseLogoPath}}}%
        }\hspace{0.5em}%
      }{%
        \quartoLicenseTextFallback%
      }%
    }%
  $endif$
  $endif$%
  $if(copyright)$\textcopyright\ $copyright.year$ $copyright.holder$$endif$%
}
